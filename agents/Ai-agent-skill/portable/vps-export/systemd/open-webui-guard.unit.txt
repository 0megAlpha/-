# /etc/systemd/system/open-webui-guard.service
[Unit]
Description=Open WebUI Container Guard
After=docker.service
Requires=docker.service

[Service]
Restart=always
RestartSec=10
ExecStart=/usr/bin/docker start -a open-webui-secure
ExecStop=/usr/bin/docker stop -t 30 open-webui-secure

[Install]
WantedBy=multi-user.target
